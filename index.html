Quick Start with OpenGradient SDK

bash 
# Install the OpenGradient SDK
pip install opengradient

# Set your wallet private key
export OG_PRIVATE_KEY="0xYourPrivateKey"

# Get OPG testnet tokens from the faucet
# https://faucet.opengradient.ai
python 
import asyncio
import opengradient as og

llm = og.LLM(private_key=os.environ.get("OG_PRIVATE_KEY"))
llm.ensure_opg_approval(opg_amount=5.0)

messages = [
    {"role": "system", "content": "You are a DeFi analyst."},
    {"role": "user", "content": "Analyze ETH price trends."}
]

result = asyncio.run(llm.chat(
    model=og.TEE_LLM.GPT_5,
    messages=messages,
    max_tokens=256
))

print(f"Response: {result.chat_output['content']}")
print(f"Payment hash: {result.payment_hash}")
python 
async def stream_inference():
    stream = await llm.chat(
        model=og.TEE_LLM.CLAUDE_SONNET_4_6,
        messages=[
            {"role": "user", 
             "content": "Explain DeFi yield farming"}
        ],
        stream=True,
        max_tokens=500
    )
    
    async for chunk in stream:
        if chunk.choices[0].delta.content:
            print(chunk.choices[0].delta.content, end="")

asyncio.run(stream_inference())
